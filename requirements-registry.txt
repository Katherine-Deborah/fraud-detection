# Session 7: model registry + staged rollout (shadow/canary harnesses) --
# isolated environment, same reason as requirements-sagemaker.txt: mlflow's
# pre-2.0 numpy/pyarrow pins conflict with feast==0.65.0's numpy>=2.0 pin in
# the main requirements.txt. This venv never imports feast/kafka code, only
# mlflow registry calls + loading/scoring the locally-trained (Session 5)
# sklearn models, so the older numpy/pyarrow here are harmless.
#
#   python -m venv .venv-registry
#   .venv-registry/Scripts/pip install -r requirements-registry.txt
#
# scikit-learn is pinned to 1.4.2 -- NOT 1.2.1 like requirements-sagemaker.txt
# -- because these scripts load the Session 5 *locally*-trained models
# (logistic_regression, random_forest, isolation_forest runs), which were
# trained under 1.4.2 (see requirements.txt). Session 6 already proved
# scikit-learn's pickled tree-node layout isn't cross-version compatible
# (1.2.1 vs 1.3+), so this venv deliberately does NOT attempt to load the
# SageMaker-trained random_forest_sagemaker model -- that version is
# registered (pointer only, no load needed) but not live-scored in Session
# 7's shadow/canary walkthrough. Same reasoning for the LSTM: it's
# registered as a pointer only (mlflow.pytorch model versions don't need to
# be loaded to create a ModelVersion from a run's artifact URI), so torch
# is intentionally not installed here either.
#
# mlflow-skinny is pinned alongside mlflow to the same version for the same
# reason documented in requirements-sagemaker.txt (an unpinned mlflow-skinny
# gets pulled in transitively at a newer version and targets a newer REST
# API the pinned mlflow==2.12.2 tracking server doesn't have).

mlflow==2.12.2
mlflow-skinny==2.12.2
scikit-learn==1.4.2
pandas==2.3.3
pyarrow==15.0.2
matplotlib==3.9.2
